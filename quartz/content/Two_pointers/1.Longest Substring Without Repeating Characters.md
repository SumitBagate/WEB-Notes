
  Naive Approach :- 
  
- Start from every index `i`.
- Keep moving `j` forward.
- Use a hash array to check whether a character has already appeared.
- Stop when you find a duplicate.
- Keep track of the maximum length.

```cpp
#include <iostream>
#include <vector>
#include <string>
using namespace std;

int longestSubstring(string s) {
    int n = s.size();
    int max_len = 0;

    for (int i = 0; i < n; i++) {
        vector<int> hash(256, 0);

        for (int j = i; j < n; j++) {

            // Character already exists
            if (hash[s[j]] == 1) {
                break;
            }

            hash[s[j]] = 1;

            // Current substring length
            max_len = max(max_len, j - i + 1);
        }
    }

    return max_len;
}

int main() {
    string s = "abcabcbb";

    cout << longestSubstring(s) << endl;

    return 0;
}
```

Time complexity : - O(n<sup>2</sup>)
Space Complexity: O(n)
