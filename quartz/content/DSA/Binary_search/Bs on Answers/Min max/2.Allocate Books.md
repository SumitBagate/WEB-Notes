class Solution {
public:

    int CntStudent(vector<int> &arr, int pages) {
        int students = 1;
        int pageStd = 0;

        for (int book : arr) {
            if (pageStd + book <= pages) {
                pageStd += book;
            }
            else {
                students++;
                pageStd = book;
            }
        }

        return students;
    }

    int findPages(vector<int> &arr, int k) {

        int n = arr.size();

        if (k > n) {
            return -1;
        }

        int low = *max_element(arr.begin(), arr.end());
        int high = accumulate(arr.begin(), arr.end(), 0);

        for (int pages = low; pages <= high; pages++) {

            if (CntStudent(arr, pages) <= k) {
                return pages;
            }
        }

        return -1;
    }
};

```cpp
class Solution {
public:

 long long countStudent(vector<int>& arr, long long pages) {
        long long pag = 0;
        long long students = 1;

        for (int pg : arr) {
            if (pag + pg <= pages) {
                pag += pg;
            }
            else {
                students++;
                pag = pg;
            }
        }

        return students;
    }



    int findPages(vector<int>& arr, int k) {

        if (arr.size() < k)
            return -1;

        long long low = *max_element(arr.begin(), arr.end());
        long long high = accumulate(arr.begin(), arr.end(), 0LL);

        while (low <= high) {

            long long mid = low + (high - low) / 2;

            long long st = countStudent(arr, mid);

            if (st > k) {
                low = mid + 1;
            }
            else {
                high = mid - 1;
            }
        }

        return low;
    }
};

```

Time complexity :- O(n); -- because for loop is used in this

## Approach summary

### Brute force

Try every possible maximum page limit from the largest book to the total pages. Greedily assign consecutive books to the current student and start a new student when the limit would be exceeded. The first feasible limit is the answer.

### Optimal: binary search on the answer

The number of students required is monotonic with respect to the page limit. A larger limit never requires more students, so binary search the page range and use the existing counting helper as the feasibility check.

**Brute-force time:** $O(n * S)$, where `S` is the sum of pages.

**Optimal time:** $O(n log S)$

**Space complexity:** $O(1)$
space complexity :- O(1); -- no extra space is used here
