## Binary search approach 


```cpp
class Solution {
public:

    bool isPossible(vector<int>& boards, int numberOfPainters, int maxTimePerPainter) {

        int paintersNeeded = 1;
        int currentPainterTime = 0;

        for (int boardTime : boards) {

            // Assign this board to the current painter
            if (currentPainterTime + boardTime <= maxTimePerPainter) {
                currentPainterTime += boardTime;
            }
            else {
                // Current painter cannot take this board
                // Assign it to a new painter
                paintersNeeded++;
                currentPainterTime = boardTime;
            }
        }

        // Can all boards be completed using available painters?
        return paintersNeeded <= numberOfPainters;
    }


    int minTime(vector<int>& boards, int numberOfPainters) {

        int numberOfBoards = boards.size();

        // More painters than boards is not useful
        if (numberOfPainters > numberOfBoards) {
            return -1;
              }

        // Minimum possible answer:
        // A painter must at least paint the largest board
        int minimumTime = *max_element(boards.begin(), boards.end());

        // Maximum possible answer:
        // One painter paints all boards
        int maximumTime = accumulate(boards.begin(), boards.end(), 0);

        while (minimumTime <= maximumTime) {

            int possibleTime = minimumTime +
                               (maximumTime - minimumTime) / 2;

            if (isPossible(boards, numberOfPainters, possibleTime)) {

                // possibleTime works.
                // Try to find an even smaller time.
                maximumTime = possibleTime - 1;
            }
            else {

                // possibleTime does not work.
                // We need more time.
                minimumTime = possibleTime + 1;
            }
        }

        return minimumTime;
    }
};
```


Time  complexity  :-  0(n) + 0(n) + 0(n log s)  two  for loop + binary  search
Space complexity :-  0(1) --  only  3 temp variable so  constant
